CREATE OR REPLACE VIEW vw_customer_order_history AS
SELECT
    c.client_id,
    c.first_name,
    c.last_name,
    c.email,

    o.order_num,
    o.quantity,
    o.status,
    o.last_modified_date,
    o.payment_method,
    o.discount,

    p.code AS product_code,
    p.description AS product_description,
    p.price

FROM client c
JOIN makes_order mo
    ON mo.client_id = c.client_id
JOIN "order" o
    ON o.order_num = mo.order_num
JOIN includes i
    ON i.order_num = o.order_num
JOIN product p
    ON p.code = i.code;
